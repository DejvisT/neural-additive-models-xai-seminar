for ($x = 1; $x -le 5; $x++) {
    for ($s = 1; $s -le 3; $s++) {

        Write-Host "Running fold $x, split $s"

        python -m neural_additive_models.nam_train `
            --logdir=./housing_default `
            --dataset_name=Housing `
            --fold_num=$x `
            --regression=True `
            --use_dnn=False `
            --num_splits=3 `
            --data_split=$s `
            --training_epochs=1000
    }
}
